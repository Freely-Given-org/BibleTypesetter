% The native cutout shapes the text but does not prohibit a page break
% through the box. Protect the affected lines (including one clearance line).
% Native cutoutcarryover handles a box extending into the next paragraph.
\let\OETMakeCutouts\makecutouts
\newcount\OETLine
\newcount\OETCut
\newcount\OETAfter
\newcount\OETUntil
\newcount\OETLinePenalty
\def\makecutouts{%
  \OETMakeCutouts
  \interlinepenalties=0
  \ifnum\@numcuts>0
    \OETLine=0
    \edef\OETPenalties{}%
    \loop\ifnum\OETLine<\@parshapelines
      \advance\OETLine by 1
      \global\OETLinePenalty=0
      {\OETCut=0
       \loop\ifnum\OETCut<\@numcuts
         \advance\OETCut by 1
         \OETAfter=\csname cut@after\the\OETCut\endcsname\relax
         \OETUntil=\csname cut@lines\the\OETCut\endcsname\relax
         \advance\OETUntil by \OETAfter
         \ifnum\OETLine>\OETAfter\ifnum\OETLine>\OETUntil\else
           \global\OETLinePenalty=10000
         \fi\fi
       \repeat}%
      \edef\OETPenalties{\OETPenalties\space\the\OETLinePenalty}%
    \repeat
    \interlinepenalties=\@parshapelines\space\OETPenalties\relax
  \fi}
